        \end{itemize}
      \end{block}

        \begin{itemize}
        \item Functional simulation
        \item Trace information

        \end{itemize}
      \end{block}   
      \begin{alertblock}{Pro / Cons}
        \begin{itemize}
        \item Pro : allow ``run to completion''
        \item Pro : allow ``plugins''

        \end{itemize}
        
      \end{alertblock}
    \end{column}
  \end{columns}
